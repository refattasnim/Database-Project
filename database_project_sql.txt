-- create enrollments table 


CREATE TABLE enrollments ( 
enrollment_id INT AUTO_INCREMENT PRIMARY KEY, 
student_id INT NOT NULL, 
course_id INT NOT NULL, 
semester VARCHAR(20) NOT NULL, 


UNIQUE (student_id, course_id, semester), 


FOREIGN KEY (student_id) REFERENCES students(student_id), 
FOREIGN KEY (course_id) REFERENCES courses(course_id) 
) ENGINE=InnoDB;




-- insert into enrollments 


INSERT INTO enrollments 
(student_id, course_id, semester) 
SELECT 
s.student_id, 
c.course_id, 
s.semester 
FROM students s 
JOIN courses c ON ( 
(s.intake_section IN ('55-1', '55-2') 
         AND c.course_code IN ('CSE 207', 'CSE 341', 'CSE 209')) 
OR (s.intake_section IN ('56-1', '56-2') 
         AND c.course_code IN ('CSE 221', 'CSE 205', 'CSE 231')) 
) 
ORDER BY s.student_id ASC, c.course_id ASC;

-- create exam room allocations table 


CREATE TABLE exam_room_allocations (
    allocation_id INT AUTO_INCREMENT PRIMARY KEY,
    schedule_id INT NOT NULL,
    room_id INT NOT NULL,
    intake_section VARCHAR(10) NOT NULL,
    allocated_seats INT NOT NULL,
    UNIQUE (schedule_id, room_id, intake_section),
    FOREIGN KEY (schedule_id)
        REFERENCES exam_schedules(schedule_id),


    FOREIGN KEY (room_id)
        REFERENCES rooms(room_id),


    CHECK (allocated_seats > 0)
) ENGINE=InnoDB;




-- insert into exam room allocations 


-- Insert dynamic room allocations


INSERT INTO exam_room_allocations
(schedule_id, room_id, intake_section, allocated_seats)


SELECT
    es.schedule_id,
    r.room_id,
    x.intake_section,
    COUNT(DISTINCT s.student_id) AS total_students


FROM (
    -- CSE 207 (55 Intake)
    SELECT 'CSE 207' AS course_code, '55-1' AS intake_section, '301' AS room_number, 'A' AS building
    UNION ALL SELECT 'CSE 207', '55-2', '302', 'A'


    -- CSE 341 (55 Intake)
    UNION ALL SELECT 'CSE 341', '55-1', '401', 'B'
    UNION ALL SELECT 'CSE 341', '55-2', '402', 'B'


    -- CSE 209 (55 Intake)
    UNION ALL SELECT 'CSE 209', '55-1', '302', 'A'
    UNION ALL SELECT 'CSE 209', '55-2', '303', 'A'


    -- CSE 221 (56 Intake)
    UNION ALL SELECT 'CSE 221', '56-1', '303', 'A'
    UNION ALL SELECT 'CSE 221', '56-2', '401', 'B'


    -- CSE 205 (56 Intake)
    UNION ALL SELECT 'CSE 205', '56-1', '301', 'A'
    UNION ALL SELECT 'CSE 205', '56-2', '302', 'A'


    -- CSE 231 (56 Intake)
    UNION ALL SELECT 'CSE 231', '56-1', '401', 'B'
    UNION ALL SELECT 'CSE 231', '56-2', '402', 'B'


) AS x


JOIN courses c
    ON c.course_code = x.course_code


JOIN exams e
    ON e.course_id = c.course_id


JOIN exam_schedules es
    ON es.exam_id = e.exam_id


JOIN rooms r
    ON r.room_number = x.room_number
    AND r.building = x.building


JOIN enrollments en
    ON en.course_id = c.course_id
    AND en.semester = e.semester


JOIN students s
    ON s.student_id = en.student_id
    AND s.intake_section = x.intake_section


GROUP BY
    es.schedule_id,
    r.room_id,
    x.intake_section


ORDER BY
    es.schedule_id ASC,
    x.intake_section ASC;

    UNIQUE (allocation_id, student_id),


    FOREIGN KEY (allocation_id)
        REFERENCES exam_room_allocations(allocation_id)
        ON DELETE CASCADE,


    FOREIGN KEY (student_id)
        REFERENCES students(student_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;


-- Assign students according to room allocation seat limits


INSERT INTO exam_room_students (allocation_id, student_id)


SELECT
    ar.allocation_id,
    sr.student_id


FROM (
    SELECT
        allocation_id,
        schedule_id,
        intake_section,
        allocated_seats,


        COALESCE(
            SUM(allocated_seats) OVER (
                PARTITION BY schedule_id, intake_section
                ORDER BY allocation_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
            ), 0
        ) AS seat_start


    FROM exam_room_allocations
) ar


JOIN (
    SELECT
        es.schedule_id,
        s.student_id,
        s.intake_section,


        ROW_NUMBER() OVER (
            PARTITION BY es.schedule_id, s.intake_section
            ORDER BY s.student_id
        ) AS student_rank


    FROM exam_schedules es
    JOIN exams e ON es.exam_id = e.exam_id
    JOIN enrollments en ON en.course_id = e.course_id
    JOIN students s
        ON s.student_id = en.student_id
        AND s.semester = en.semester
    WHERE e.semester = en.semester
) sr


ON sr.schedule_id = ar.schedule_id
AND sr.intake_section = ar.intake_section


WHERE sr.student_rank > ar.seat_start
AND sr.student_rank <= ar.seat_start + ar.allocated_seats;

-- room insert trigger


DELIMITER $$


CREATE TRIGGER validate_room_insert
BEFORE INSERT ON exam_room_allocations
FOR EACH ROW
BEGIN


    DECLARE room_capacity INT;
    DECLARE existing_seats INT;


    SELECT capacity
    INTO room_capacity
    FROM rooms
    WHERE room_id = NEW.room_id;


    SELECT COALESCE(SUM(allocated_seats), 0)
    INTO existing_seats
    FROM exam_room_allocations
    WHERE room_id = NEW.room_id
      AND schedule_id = NEW.schedule_id;


    -- Check room capacity
    IF NEW.allocated_seats <= 0
       OR existing_seats + NEW.allocated_seats > room_capacity THEN


        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Room capacity exceeded! Allocation cannot be inserted.';


    END IF;


    -- Check room time conflict
    IF EXISTS (
        SELECT 1
        FROM exam_room_allocations a


        JOIN exam_schedules es
            ON a.schedule_id = es.schedule_id


        JOIN exam_schedules new_es
            ON new_es.schedule_id = NEW.schedule_id


        WHERE a.room_id = NEW.room_id
          AND es.schedule_id <> NEW.schedule_id


          AND es.exam_date = new_es.exam_date


          AND new_es.start_time < es.end_time
          AND new_es.end_time > es.start_time


    ) THEN


        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Room is already booked! Allocation cannot be inserted.';


    END IF;


END$$


DELIMITER ;

-- room update trigger


DELIMITER $$


CREATE TRIGGER validate_room_update
BEFORE UPDATE ON exam_room_allocations
FOR EACH ROW
BEGIN


    DECLARE room_capacity INT;
    DECLARE existing_seats INT;
    DECLARE assigned_students INT;


    SELECT capacity
    INTO room_capacity
    FROM rooms
    WHERE room_id = NEW.room_id;


    SELECT COALESCE(SUM(allocated_seats), 0)
    INTO existing_seats
    FROM exam_room_allocations
    WHERE room_id = NEW.room_id
      AND schedule_id = NEW.schedule_id
      AND allocation_id <> OLD.allocation_id;


    -- Check already assigned students
    SELECT COUNT(*)
    INTO assigned_students
    FROM exam_room_students
    WHERE allocation_id = OLD.allocation_id;


    -- Prevent seats from becoming less than assigned students
    IF NEW.allocated_seats < assigned_students THEN


        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Allocated seats cannot be less than assigned students!';


    END IF;


    -- Check room capacity
    IF NEW.allocated_seats <= 0
       OR existing_seats + NEW.allocated_seats > room_capacity THEN


        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Room capacity exceeded! Allocation cannot be updated.';


    END IF;


    -- Check room time conflict
    IF EXISTS (
        SELECT 1
        FROM exam_room_allocations a


        JOIN exam_schedules es
            ON a.schedule_id = es.schedule_id


        JOIN exam_schedules new_es
            ON new_es.schedule_id = NEW.schedule_id


        WHERE a.room_id = NEW.room_id
          AND a.allocation_id <> OLD.allocation_id
          AND a.schedule_id <> NEW.schedule_id
          AND es.exam_date = new_es.exam_date
          AND new_es.start_time < es.end_time
          AND new_es.end_time > es.start_time


    ) THEN


        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Room is already booked! Allocation cannot be updated.';


    END IF;


END$$
